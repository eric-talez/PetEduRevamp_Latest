export const notFoundHandler = (req: Request, res: Response) => {
  const response = createErrorResponse(
    ApiErrorCode.RESOURCE_NOT_FOUND,
    `요청하신 경로 ${req.originalUrl}을 찾을 수 없습니다.`
  );
  res.status(HTTP_STATUS.NOT_FOUND).json(response);
};
