function renderTemplate(text: string, vars: Record<string, any> = {}): string {
  return text.replace(/\{\{\s*(\w+)\s*\}\}/g, (_m, name) => {
    const v = vars[name];
    return v === undefined || v === null ? "" : String(v);
  });
}
